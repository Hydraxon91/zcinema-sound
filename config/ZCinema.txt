# ---- SECTION 1: Ceiling / headroom -----------------------------------------
#  Lower = quieter max = more usable slider travel. Try -6 .. -13 dB.
#  Also acts as headroom for the boosts below (prevents clipping).
Preamp: -8 dB

# ---- SECTION 2: Bass  (TruBass-like)  gain -12 .. +12 dB ---------------------
#  Broad peaking bands (a true shelf is not used here because some Equalizer
#  APO builds ignore the LSC/HSC shelf types).
Filter: ON  PK  Fc 100 Hz Gain  6 dB Q 0.70    # main bass band
Filter: ON  PK  Fc  45 Hz Gain  4 dB Q 1.20    # sub-bass weight
Filter: ON  PK  Fc 250 Hz Gain -2 dB Q 1.00    # de-mud

# ---- SECTION 3: Treble  gain -12 .. +12 dB ----------------------------------
Filter: ON  PK  Fc 8000 Hz Gain 4 dB Q 0.70

# ---- SECTION 4: Dialogue clarity  (SRS DialogClarity-like) ------------------
Filter: ON  PK  Fc 3000 Hz Gain 3 dB Q 1.50

# ---- SECTION 5: Stereo width / TruSurround-like  (g = 0.05 .. 0.20) ---------
#  Negative crossfeed widens the image (added as a negative factor; Equalizer
#  APO's Copy grammar adds terms, so use "+-0.10" not "-0.10").
#  Comment both lines for strict mono. Lower g toward 0.05 if mono collapses.
Copy: L=L+-0.10*R
Copy: R=R+-0.10*L
